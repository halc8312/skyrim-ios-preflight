cmake_minimum_required(VERSION 3.20)
project(SkyrimProbeCore VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(guest_memory STATIC native/guest_memory.c)
target_include_directories(guest_memory PUBLIC native)
if(MSVC)
  target_compile_options(guest_memory PRIVATE /W4 /WX)
else()
  target_compile_options(guest_memory PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(test_guest_memory tests/native/test_guest_memory.c)
target_link_libraries(test_guest_memory PRIVATE guest_memory)
add_test(NAME guest_memory COMMAND test_guest_memory)
