cmake_minimum_required(VERSION 3.20)
project(SkyrimEnbEffectExec LANGUAGES C CXX)

# Runs authored ENB-style effects on a real D3D9 device: the port's pinned native DXVK on a Vulkan driver
# (software lavapipe is enough for pixel checks) with HLSL compiled by libvkd3d-shader 2.0.
set(SKYDREAM_SOURCE "" CACHE PATH "Pinned Skydream checkout (commit ecf814f58292d70ad5cfb19a5c61b5d8d8006a4a)")
set(ENB_VKD3D_PREFIX "" CACHE PATH "Prefix holding include/vkd3d/vkd3d_shader.h (or the source include dir) and lib/libvkd3d-shader.so")
set(ENB_VKD3D_INCLUDE "" CACHE PATH "Directory containing vkd3d_shader.h")
set(ENB_VKD3D_LIBRARY "" CACHE FILEPATH "libvkd3d-shader shared library")
if(NOT EXISTS "${SKYDREAM_SOURCE}/third_party/dxvk/src/d3d9/d3d9_device.cpp")
    message(FATAL_ERROR "Set SKYDREAM_SOURCE to the pinned Skydream checkout")
endif()
if(NOT EXISTS "${ENB_VKD3D_INCLUDE}/vkd3d_shader.h" OR NOT EXISTS "${ENB_VKD3D_LIBRARY}")
    message(FATAL_ERROR "Set ENB_VKD3D_INCLUDE and ENB_VKD3D_LIBRARY (vkd3d 2.0, see README.md)")
endif()
find_package(Threads REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
if(NOT TARGET SDL2)
    add_library(SDL2 ALIAS SDL2::SDL2)
endif()
set(dxvk "${SKYDREAM_SOURCE}/third_party/dxvk")
add_library(Vulkan::Headers INTERFACE IMPORTED)
set_target_properties(Vulkan::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${dxvk}/include")
add_subdirectory("${dxvk}" dxvk)

set(port "${CMAKE_CURRENT_SOURCE_DIR}/../../ios/runtime-port")
set(CMAKE_C_STANDARD 11)
add_executable(enb_exec_test enb_exec_test.cpp "${port}/enb_exec.cpp" "${port}/enb_runtime.cpp"
    "${port}/enb_fx.c" "${port}/enb_ini.c" "${port}/enb_preset.c" "${port}/mod_plugins.c")
target_compile_features(enb_exec_test PRIVATE cxx_std_17)
target_compile_definitions(enb_exec_test PRIVATE DXVK_NATIVE NOMINMAX _WIN32_WINNT=0x0a00)
target_include_directories(enb_exec_test PRIVATE "${port}" "${ENB_VKD3D_INCLUDE}")
target_link_libraries(enb_exec_test PRIVATE dxvk_d3d9 SDL2 "${ENB_VKD3D_LIBRARY}" m)
enable_testing()
add_test(NAME enb_exec COMMAND enb_exec_test)
