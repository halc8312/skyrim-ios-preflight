cmake_minimum_required(VERSION 3.20)
project(SkyrimD3D9PostprocessProbe LANGUAGES C CXX)

set(SKYDREAM_SOURCE "" CACHE PATH "Skydream checkout or pinned source archive root")
set(POSTFX_SDL_INCLUDE "" CACHE PATH "Optional SDL2 include directory containing SDL.h")
set(POSTFX_SDL_LIBRARY "" CACHE FILEPATH "Matching SDL2 library when using an explicit include directory")

add_executable(d3d9_postprocess probe.cpp postprocess.cpp depth_input.cpp)
target_compile_features(d3d9_postprocess PRIVATE cxx_std_17)
target_compile_definitions(d3d9_postprocess PRIVATE NOMINMAX)
if(WIN32 AND NOT SKYDREAM_SOURCE)
    target_link_libraries(d3d9_postprocess PRIVATE d3d9 user32)
else()
    if(NOT EXISTS "${SKYDREAM_SOURCE}/third_party/dxvk/src/d3d9/d3d9_device.cpp")
        message(FATAL_ERROR "Set SKYDREAM_SOURCE to the project's pinned Skydream source")
    endif()
    find_package(Threads REQUIRED)
    if(POSTFX_SDL_INCLUDE OR POSTFX_SDL_LIBRARY)
        if(NOT EXISTS "${POSTFX_SDL_INCLUDE}/SDL.h" OR NOT EXISTS "${POSTFX_SDL_LIBRARY}")
            message(FATAL_ERROR "Supply paths for both the SDL2 headers and library")
        endif()
        add_library(SDL2 INTERFACE)
        # Skydream includes SDL headers with and without the SDL2/ prefix.
        get_filename_component(sdl_parent "${POSTFX_SDL_INCLUDE}" DIRECTORY)
        target_include_directories(SDL2 SYSTEM INTERFACE "${POSTFX_SDL_INCLUDE}" "${sdl_parent}")
        target_link_libraries(SDL2 INTERFACE "${POSTFX_SDL_LIBRARY}")
    else()
        find_package(SDL2 CONFIG REQUIRED)
        if(NOT TARGET SDL2)
            add_library(SDL2 ALIAS SDL2::SDL2)
        endif()
    endif()
    set(dxvk "${SKYDREAM_SOURCE}/third_party/dxvk")
    add_library(Vulkan::Headers INTERFACE IMPORTED)
    set_target_properties(Vulkan::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${dxvk}/include")
    add_subdirectory("${dxvk}" dxvk)
    target_compile_definitions(d3d9_postprocess PRIVATE POSTFX_NATIVE_DXVK DXVK_NATIVE _WIN32_WINNT=0x0a00)
    target_link_libraries(d3d9_postprocess PRIVATE dxvk_d3d9 SDL2)
endif()

enable_testing()
add_test(NAME d3d9_postprocess_pixels_and_state COMMAND d3d9_postprocess)
add_test(NAME d3d9_postprocess_depth_input COMMAND d3d9_postprocess --depth-input)
add_test(NAME d3d9_postprocess_underwater_fog COMMAND d3d9_postprocess --underwater-fog)
set_tests_properties(d3d9_postprocess_depth_input d3d9_postprocess_underwater_fog PROPERTIES SKIP_RETURN_CODE 77)
