cmake_minimum_required(VERSION 3.20)
project(SkyrimPresentGradeProbe LANGUAGES CXX)
enable_testing()

set(GRADE_DXVK "" CACHE PATH "Pinned Skydream third_party/dxvk")
set(GRADE_GENERATED "" CACHE PATH "New directory produced by prepare.py")
set(GRADE_DXVK_LIBRARY "" CACHE FILEPATH "Existing native pinned DXVK D3D9 library")
option(GRADE_BUILD_STAGED_HOST "Build a separate DXVK library with the opt-in host integration" OFF)
set(GRADE_SDL_INCLUDE "" CACHE PATH "Matching SDL2 include directory containing SDL.h")
set(GRADE_SDL_LIBRARY "" CACHE FILEPATH "Matching native SDL2 library")
foreach(required IN ITEMS "${GRADE_DXVK}/src/dxvk/dxvk_swapchain_blitter.cpp" "${GRADE_GENERATED}/grade_trial_blitter.cpp"
                          "${GRADE_SDL_INCLUDE}/SDL.h" "${GRADE_SDL_LIBRARY}")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing required input: ${required}")
  endif()
endforeach()
get_filename_component(sdl_parent "${GRADE_SDL_INCLUDE}" DIRECTORY)
if(GRADE_BUILD_STAGED_HOST)
  enable_language(C)
  find_package(Threads REQUIRED)
  add_library(SDL2 INTERFACE)
  target_include_directories(SDL2 SYSTEM INTERFACE "${GRADE_SDL_INCLUDE}" "${sdl_parent}")
  target_link_libraries(SDL2 INTERFACE "${GRADE_SDL_LIBRARY}")
  add_library(Vulkan::Headers INTERFACE IMPORTED)
  set_target_properties(Vulkan::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${GRADE_DXVK}/include")
  add_subdirectory("${GRADE_DXVK}" dxvk)
  include(staged_host.cmake)
  skyrim_trial_present_grade(dxvk_d3d9 "${GRADE_DXVK}" "${GRADE_GENERATED}")
  set(GRADE_DXVK_LIBRARY dxvk_d3d9)
elseif(NOT EXISTS "${GRADE_DXVK_LIBRARY}")
  message(FATAL_ERROR "Supply a pinned native DXVK library, or enable GRADE_BUILD_STAGED_HOST")
endif()
add_executable(present_grade probe.cpp "${GRADE_GENERATED}/grade_trial_blitter.cpp")
target_compile_features(present_grade PRIVATE cxx_std_17)
target_compile_definitions(present_grade PRIVATE DXVK_NATIVE DXVK_WSI_SDL2 SDL_MAIN_HANDLED NOMINMAX _WIN32_WINNT=0xa00)
target_include_directories(present_grade PRIVATE "${GRADE_GENERATED}")
target_include_directories(present_grade SYSTEM PRIVATE "${GRADE_DXVK}/include/native/directx" "${GRADE_DXVK}/include/native/windows"
  "${GRADE_DXVK}/include/native" "${GRADE_DXVK}/src/d3d9" "${GRADE_DXVK}/src/dxvk" "${GRADE_DXVK}/src"
  "${GRADE_DXVK}/generated" "${GRADE_DXVK}/include" "${GRADE_SDL_INCLUDE}" "${sdl_parent}")
target_link_libraries(present_grade PRIVATE "${GRADE_DXVK_LIBRARY}" "${GRADE_SDL_LIBRARY}")
add_test(NAME present_grade_gpu_pixels_state_and_draws COMMAND present_grade)
if(GRADE_BUILD_STAGED_HOST)
  add_test(NAME present_grade_host_selection COMMAND present_grade --staged-check)
endif()
