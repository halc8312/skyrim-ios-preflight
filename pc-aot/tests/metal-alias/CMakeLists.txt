cmake_minimum_required(VERSION 3.24)
project(SamplerAliasRepro LANGUAGES CXX)
if(NOT SPIRV_CROSS_SOURCE)
  message(FATAL_ERROR "SPIRV_CROSS_SOURCE is required")
endif()
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
add_executable(sampler-alias alias.cpp)
if(SPIRV_CROSS_LIB_DIR)
  # Local reproduction can reuse a just-built compiler instead of rebuilding it.
  target_include_directories(sampler-alias PRIVATE "${SPIRV_CROSS_SOURCE}")
  foreach(name spirv-cross-msl spirv-cross-glsl spirv-cross-core)
    find_library(${name}_LIB NAMES ${name} PATHS "${SPIRV_CROSS_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(sampler-alias PRIVATE "${${name}_LIB}")
  endforeach()
else()
  add_subdirectory("${SPIRV_CROSS_SOURCE}" cross)
  target_link_libraries(sampler-alias PRIVATE spirv-cross-msl)
endif()
target_compile_features(sampler-alias PRIVATE cxx_std_17)
